**Step 3: Implement AuditDaemon**

```python
"""
AuditDaemon — subprocess lifecycle manager for the audit log process.

Spawns argus.security.audit.log_process as a child process with a Unix socket
scoped per-run. Uses socket.connect() probe for readiness (not file existence).
Fail-closed: raises AuditUnavailableError if daemon doesn't start.
"""
from __future__ import annotations

import os
import socket
import subprocess
import sys
import time
from typing import Self

from argus.security.exceptions import AuditUnavailableError


class AuditDaemon:
    """Manages the audit logger subprocess lifecycle.

    Usage:
        with AuditDaemon(socket_path, log_path) as daemon:
            audit_logger = AuditLogger(socket_path)
            # ... use audit_logger ...
        # daemon is stopped and cleaned up
    """

    def __init__(
        self,
        socket_path: str,
        log_path: str,
        timeout_s: float = 5.0,
    ) -> None:
        self._socket_path = socket_path
        self._log_path = log_path
        self._timeout_s = timeout_s
        self._proc: subprocess.Popen | None = None

    def start(self) -> None:
        """Spawn the daemon and wait until the socket is ready.

        Raises:
            AuditUnavailableError: If daemon fails to start within timeout.
        """
        self._proc = subprocess.Popen(
            [
                sys.executable, "-m", "argus.security.audit.log_process",
                self._socket_path, self._log_path,
            ],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
        )
        self._wait_for_socket_ready()

    def stop(self) -> None:
        """Terminate the daemon. Idempotent — safe to call multiple times."""
        if self._proc is None:
            return
        try:
            self._proc.terminate()
            self._proc.wait(timeout=2)
        except (ProcessLookupError, OSError):
            pass
        except subprocess.TimeoutExpired:
            self._proc.kill()
            self._proc.wait(timeout=1)
        finally:
            self._proc = None
            # Clean up stale socket file
            if os.path.exists(self._socket_path):
                try:
                    os.unlink(self._socket_path)
                except OSError:
                    pass

    @property
    def pid(self) -> int | None:
        return self._proc.pid if self._proc else None

    def is_alive(self) -> bool:
        if self._proc is None:
            return False
        return self._proc.poll() is None

    def _wait_for_socket_ready(self) -> None:
        """Probe with actual socket.connect() — not just file existence."""
        deadline = time.monotonic() + self._timeout_s
        while time.monotonic() < deadline:
            # Check if process died
            if self._proc and self._proc.poll() is not None:
                raise AuditUnavailableError(
                    f"Audit daemon exited with code {self._proc.returncode}"
                )
            try:
                sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
                sock.connect(self._socket_path)
                sock.close()
                return
            except (ConnectionRefusedError, FileNotFoundError, OSError):
                time.sleep(0.05)
        self.stop()
        raise AuditUnavailableError(
            f"Audit daemon socket not ready at {self._socket_path} "
            f"after {self._timeout_s}s"
        )

    def __enter__(self) -> Self:
        self.start()
        return self

    def __exit__(self, *exc: object) -> None:
        self.stop()
```

```python
"""argus run — full Argus runtime execution."""
from __future__ import annotations
import asyncio
import os
from pathlib import Path

import typer

app = typer.Typer()


@app.command()
def run_command(
    config: Path = typer.Option(Path("argus.yaml"), "--config", help="Path to argus.yaml"),
    task: str = typer.Option("", "--task", help="Task string to run"),
    trace_dir: Path = typer.Option(Path("./runs"), "--trace-dir", help="Directory for trace output"),
) -> None:
    """Execute an agent through the full Argus runtime."""
    if not os.environ.get("ANTHROPIC_API_KEY"):
        typer.echo("Error: ANTHROPIC_API_KEY environment variable is not set.", err=True)
        raise typer.Exit(code=2)

    if not config.exists():
        typer.echo(f"Error: Config file not found: {config}", err=True)
        raise typer.Exit(code=2)

    asyncio.run(_run_async(config, task, trace_dir))


async def _run_async(config: Path, task: str, trace_dir: Path) -> None:
    """Wire and run the full Argus runtime stack."""
    from argus.llm.config import load_config
    from argus.llm.tracker import SpendTracker
    from argus.llm.router import LLMRouter
    from argus.engine.machine import StateMachine
    from argus.engine.states import RunContext
    from argus.security.gateway import SecurityGateway, GatewayConfig
    from argus.security.audit.logger import AuditLogger
    from argus.security.audit.daemon import AuditDaemon
    from argus.security.redactor.redactor import SecretRedactor
    from argus.memory.manager import MemoryManager, MemoryConfig
    from argus.observability.manager import ObservabilityManager, ObsConfig

    trace_dir.mkdir(parents=True, exist_ok=True)

    obs = ObservabilityManager(ObsConfig(
        trace_path=trace_dir / "trace.jsonl",
        security_stream_path=trace_dir / "security.jsonl",
        enabled=True,
    ))

    model_config = load_config(config)
    tracker = SpendTracker(model_config.spend)
    router = LLMRouter(config=model_config, tracker=tracker, obs=obs, redactor=SecretRedactor())

    # Spawn audit daemon — fail-closed if it doesn't start
    socket_path = str(trace_dir / "audit.sock")
    log_path = str(trace_dir / "audit.jsonl")
    daemon = AuditDaemon(socket_path=socket_path, log_path=log_path)
    daemon.start()

    try:
        audit_logger = AuditLogger(socket_path)
        gateway = SecurityGateway(config=GatewayConfig(), audit_logger=audit_logger, obs=obs)

        memory = MemoryManager(MemoryConfig(db_path=trace_dir / "memory.db"))
        await memory.connect()

        try:
            sm = StateMachine(
                gateway=gateway,
                cost_hook=tracker.over_budget,
                llm_callable=router,
                store=memory.session("run-session"),
                obs=obs,
            )
            ctx = RunContext(
                task_id="cli-run",
                task_input={"goal": task if task else "demo task"},
            )
            result = await sm.run(ctx)
            result.cost_breakdown = tracker.entries()
            obs.on_run_complete(result)
            obs.flush()

            if result.success:
                typer.echo(f"Run complete. Trace written to {trace_dir / 'trace.jsonl'}")
                raise typer.Exit(code=0)
            else:
                typer.echo(
                    f"Run failed: {result.error or 'security violation or cost abort'}",
                    err=True,
                )
                raise typer.Exit(code=1)
        finally:
            await memory.close()
    finally:
        daemon.stop()
```

**Step 1: Update scan to use lifecycle for skill execution**

The scan command currently imports `security_audit.run` directly. Wire it through lifecycle instead. However, `argus scan` is a static analysis tool that doesn't need full lifecycle (no sandbox, no subprocess execution). The lifecycle manager's `_do_execute` runs skills as subprocesses which doesn't match scan's use case.

**Design decision:** For scan, we wire lifecycle stages INSTALL → VERIFY → REPORT but skip SANDBOX/EXECUTE/MONITOR since scan is a static analysis — the skill's `run()` is called in-process, not via subprocess. Add a `verify_only` method to SkillLifecycleManager that runs INSTALL → VERIFY and returns the manifest, confirming integrity.

Add to `argus/skills/lifecycle.py`:

```python
def verify_skill(self, skill_dir: Path) -> SkillManifest:
    """Run INSTALL + VERIFY stages only. Returns validated manifest.

    Use for static analysis where full lifecycle (subprocess execution) is not needed.
    """
    manifest = self._do_install(skill_dir)
    self._do_verify(skill_dir, manifest)
    return manifest
```

Then update `argus/cli/scan.py` to verify before running:

```python
"""argus scan — static security analysis."""
from __future__ import annotations
import dataclasses
import json
from enum import Enum
from pathlib import Path

import typer

app = typer.Typer()


class OutputFormat(str, Enum):
    text = "text"
    json = "json"


@app.command()
def scan_command(
    target: Path = typer.Argument(..., help="Path to skill directory or agent config YAML"),
    fmt: OutputFormat = typer.Option(OutputFormat.text, "--format", help="Output format"),
) -> None:
    """Static security scan of a skill manifest or agent config."""
    if not target.exists():
        typer.echo(f"Error: Target not found: {target}", err=True)
        raise typer.Exit(code=2)

    from argus.skills.lifecycle import SkillLifecycleManager
    from argus.skills.security_audit import run as audit_run
    from argus.cli.output import print_violation_table

    # Verify skill integrity before scanning (INSTALL + VERIFY stages)
    skill_yaml = target / "skill.yaml" if target.is_dir() else None
    if skill_yaml and skill_yaml.exists():
        lifecycle = SkillLifecycleManager()
        lifecycle.verify_skill(target)

    report = audit_run(target)

    if fmt == OutputFormat.json:
        typer.echo(json.dumps(dataclasses.asdict(report), indent=2))
    else:
        if report.findings:
            violations = [
                {
                    "type": f.rule_id,
                    "severity": str(f.severity),
                    "details": f.message,
                }
                for f in report.findings
            ]
            print_violation_table(violations)
        else:
            typer.echo("No issues found.")

    raise typer.Exit(code=0 if report.passed else 1)
```

```python
"""
LangChain adapter — wraps LangChain tools with Argus security enforcement.

Uses a proxy/wrapper pattern around tool.invoke() — NOT a BaseCallbackHandler.
LangChain v1 backgrounds callbacks, so security violations would be silently
swallowed. The proxy pattern intercepts synchronously and fail-closed.

Usage:
    from argus.adapters.langchain import wrap_tools
    from argus.security.gateway import SecurityGateway, GatewayConfig
    from argus.security.audit.logger import AuditLogger

    gateway = SecurityGateway(config=GatewayConfig(), audit_logger=audit_logger)
    safe_tools = wrap_tools(tools, gateway=gateway, agent_role="my_agent")
    agent = create_react_agent(llm, safe_tools)
"""
from __future__ import annotations

from typing import Any

from argus.security.gateway import SecurityGateway


class ArgusToolWrapper:
    """Wraps a LangChain tool with Argus security gates.

    Intercepts invoke() to run pre_tool_call and post_tool_call through
    the SecurityGateway. Exceptions propagate synchronously — never backgrounded.
    """

    def __init__(
        self,
        tool: Any,
        gateway: SecurityGateway,
        agent_role: str = "default",
    ) -> None:
        self._tool = tool
        self._gateway = gateway
        self._agent_role = agent_role
        self.name = tool.name

    def invoke(self, input: dict[str, Any] | str, **kwargs: Any) -> str:
        """Execute tool with Argus security enforcement."""
        tool_input = input if isinstance(input, dict) else {"input": input}

        # Pre-tool security gates (permission check, audit)
        self._gateway.pre_tool_call(self._agent_role, self.name, tool_input)

        # Execute the actual tool
        raw_output = self._tool.invoke(input, **kwargs)
        output_str = str(raw_output) if not isinstance(raw_output, str) else raw_output

        # Post-tool security gates (injection scan, redaction, egress, audit)
        clean_output = self._gateway.post_tool_call(output_str)

        return clean_output

    def __getattr__(self, name: str) -> Any:
        """Proxy all other attributes to the wrapped tool."""
        return getattr(self._tool, name)


def wrap_tools(
    tools: list[Any],
    gateway: SecurityGateway,
    agent_role: str = "default",
) -> list[ArgusToolWrapper]:
    """Wrap a list of LangChain tools with Argus security enforcement.

    Args:
        tools: List of LangChain BaseTool instances.
        gateway: Configured SecurityGateway instance.
        agent_role: Role identifier for permission enforcement.

    Returns:
        List of ArgusToolWrapper instances (drop-in replacements).
    """
    return [
        ArgusToolWrapper(tool=t, gateway=gateway, agent_role=agent_role)
        for t in tools
    ]
```
